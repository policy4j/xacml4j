# The call: one headless client session with one allowed research tool.
# prompt-openai.txt tells the session to call
# mcp__deep-research-mcp__deep_research once with the query, to call
# nothing else and not to retry.
CLAUDE_CODE_MCP_TOOL_IDLE_TIMEOUT=3600000 MAX_MCP_OUTPUT_TOKENS=200000 \
rtk proxy claude -p --model haiku --output-format stream-json --verbose \
  --allowedTools "mcp__deep-research-mcp__deep_research" "ToolSearch" \
  < "$T/prompt-openai.txt" > "$T/openai.stream.jsonl" 2> "$T/openai.stderr.txt"

# Query: the same as in builds/022-api-route-trial-openai/command.txt.
